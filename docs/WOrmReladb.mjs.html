/**
 * 操作關聯式資料庫
 *
 * 注意: 各model內id欄位不是主鍵(primary key)時需要強制更改成為主鍵，否則sequelize無法匯入
 *
 * 注意: opt.url、opt.db、opt.cl、opt.storage(儲存位址名稱與路徑)含孤立代理碼元(UTF-16代理對被拆開後只剩一半，String.prototype.isWellFormed()為false)者，
 * 建構時即拋出invalid opt.<名>: string contains lone surrogate(不回顯值)：UTF-8轉換會把孤立代理碼元改寫為U+FFFD，不同名稱將指向同一儲存
 *
 * 注意: 寫入與主鍵查詢之字串須良構(第66項)：主鍵不得含孤立代理碼元或NUL，insert、insertBulk、save之數據字串值與鍵名(含巢狀、陣列元素)不得含孤立代理碼元(值之NUL不拒)；
 * 不合格者insert與insertBulk整批reject、save與del該筆ok為0附err、selectByPk回傳null，詳見各函數說明與README。
 * 註: mssql預設排序規則(如SQL_Latin1_General_CP1_CI_AS)下字串相等由排序規則決定，良構字串亦可能相等('a'、'a'+emoji、'a'+U+FFFD)，屬資料庫語意不在檢核範圍，
 * 需精確比對時請以二進位排序規則(如Latin1_General_100_BIN2)建表
 *
 * 注意: 各函數reject之值、error事件之錯誤訊息與逐筆結果之err，凡來自後端、驅動、model檔或呼叫端數據之getter(外來值)者，其中之憑證(密碼至少4字元、帳號:密碼、連線字串之帳密、PRAGMA key之值)以***遮罩：
 * 含憑證者為遮罩後之副本(instanceof同類、其餘屬性保留)，無憑證者為原物件；單獨帳號與儲存路徑不遮(dspec〈診斷輸出與遮罩〉Q3)。本套件以固定詞彙自行產生之訊息(如invalid find: …、can not import model: …)原樣交出
 *
 * 注意: 連線被後端或網路於空檔(無進行中語句，如伺服器重啟、連線被KILL、閒置逾時、網路斷線)中斷時，驅動於連線物件發出之'error'事件由sequelize之連線管理接手(連線建立後即永久掛監聽並自連線池移除該連線)，不使行程終止、不輸出後端錯誤原文；
 * 以該連線進行之操作(含以失效之transaction進行者)以其既有之失敗通道回報(reject)，其後之操作自連線池取得新連線。本套件不自建驅動連線、亦不掛驅動之監聽(R18)；此行為依賴sequelize 6於mssql、mysql、mariadb、postgres之連線管理(sqlite為本機檔案，無此種連線)
 *
 * 注意: sqlite之opt.storage含NUL字元、或恰為':memory:'者，建構時即拋出invalid opt.storage: <原因>(不回顯值；第213項, R20)：sqlite3於NUL截斷檔名，不同名稱(如a\0b、a\0c)會指向同一檔；
 * opt.storage經path.resolve，故':memory:'不是記憶體資料庫(POSIX為cwd下名為:memory:之檔案，win32為非法路徑)，且各函數每次初始化與關閉，記憶體資料庫無法跨呼叫保留，故不支援。僅sqlite檢查，其他方言不使用opt.storage
 *
 * 注意: sqlite之opt.storage無法開啟(SQLITE_CANTOPEN: 路徑為資料夾、檔名超長、無權限、路徑非法等)時，需連線之各函數(select、selectByPk、insert、insertBulk、save、del、delAll、createStorage、genTransaction與init回傳之close)以既有失敗通道回報(reject；createStorage為can not import model訊息)，不會永不settle(第210項, R19)：
 * sequelize 6把開啟失敗之Database留在連線表，其close()永不回呼，本套件於關閉前先移除未開啟(Database.open嚴格為false)者(dropUnopenedConn)。reject之訊息不帶連線憑證。
 *
 * @class
 * @param {Object} [opt={}] 輸入設定物件，預設{}
 * @param {String} [opt.url='mssql://username:password@localhost:1433'] 輸入連接資料庫字串，資料庫可選'mssql'、'sqlite'、'mysql'、'mariadb'、'postgres'，預設'mssql://username:password@localhost:1433'。格式不合者建構時拋出Error(訊息前綴如invalid host in opt.url)，訊息內回顯之url只含scheme：scheme為上列五者之一時為'<scheme>://***'，其餘(含無://者)整段為***，不回顯主機、帳號與密碼(wsemi之maskDiagUrl)
 * @param {String} [opt.storage='./worm.db'] 輸入為sqlite時，資料庫檔案所在位置字串，預設'./worm.db'。須為檔案路徑：含NUL字元(\0)、或恰為':memory:'者建構時拋出Error(訊息invalid opt.storage: string contains NUL、invalid opt.storage: must be a file path, an in-memory database is not supported，不回顯值)；含孤立代理碼元者亦拋出(invalid opt.storage: string contains lone surrogate)
 * @param {Boolean} [opt.useEncryption=false] 輸入為sqlite時是否使用加密套件[@journeyapps/sqlcipher]，型別為布林值，預設false，若是的話因依賴預設不安裝，得自行人工安裝
 * @param {String} [opt.db='worm'] 輸入使用資料庫名稱字串，預設'worm'
 * @param {String} [opt.cl='test'] 輸入使用資料表名稱字串，預設'test'
 * @param {String} [opt.fdModels='./models'] 輸入資料表設定檔所在資料夾字串，預設'./models'
 * @param {String} [opt.modelType='js'] 輸入資料表設定檔類型字串，可有'js'、'json'，預設'js'
 * @param {Boolean} [opt.logging=false] 輸入是否紀錄實際執行的sql指令布林值，預設false。為true時預設只輸出敘述種類(如SELECT)，不輸出sql全文：select、selectByPk、del、delAll之where條件值為內嵌於sql內(含主鍵與欄位值)，需搭配opt.logDetail或opt.funLog才取得sql全文，有洩漏個資之風險，僅供除錯使用
 * @param {Boolean} [opt.logDetail=false] 輸入是否於console輸出遮罩後之細節布林值，預設false，型別錯誤者視同false。診斷輸出(console)預設只含固定標籤與不含個資之摘要(不含資料值、id、後端錯誤原文、路徑)；為true時console輸出(label, 細節)(訂閱函數之錯誤、model載入失敗之原文、opt.logging之sql全文如console.log('sql', sql全文)等)，有洩漏個資之風險，僅供除錯使用。連線字串、帳號、密碼與sqlite檔案路徑於細節皆以***遮罩。出口、遮罩與選項語意同wsemi之getDiag(規則見wsemi之R-DIAG-01)
 * @param {Function} [opt.funLog=null] 輸入診斷輸出函數(label, safe, detail)，預設無，型別錯誤者視同未給。給予時診斷輸出只交給它而不寫console：label為固定標籤字串，safe為不含個資之摘要物件(可能為undefined；鍵限stmt、mode、cl、name、type、errType、errCode)，detail為完整細節之遮罩拷貝(可能無此引數；連線字串、帳號、密碼與sqlite檔案路徑已以***遮罩；拷貝之形態與上限依wsemi之maskDiagCopy，不執行使用端程式)，供使用端遮罩後轉送。函數拋錯或回傳reject之promise皆被吞下，不影響各操作之結果
 * @param {String} [opt.pk='id'] 輸入數據主鍵字串，預設'id'
 * @param {Boolean} [opt.autoGenPk=true] 輸入若數據pk(id)欄位沒給時則自動給予隨機uuid，型別為布林值，預設true
 * @param {Boolean} [opt.useStable=true] 輸入是否使用穩定模式，使用佇列管理同時只能進行一種操作故會犧牲效能，sqlite需開啟穩定模式才不會有非預期錯誤，型別為布林值，預設true
 * @returns {Object} 回傳操作資料庫物件，各事件功能詳見說明
 */
function WOrmReladb(opt = {}) {
    let ss
    let u
    let sequelize = null


    //default
    if (!isestr(opt.url)) {
        opt.url = 'mssql://username:password@localhost:1433'
    }
    if (!isestr(opt.db)) {
        opt.db = 'worm'
    }
    if (!isestr(opt.cl)) {
        opt.cl = 'test'
    }
    if (!isestr(opt.fdModels)) {
        opt.fdModels = './models'
    }
    if (!isestr(opt.modelType)) {
        opt.modelType = 'js'
    }
    opt.logging = (opt.logging === true)
    if (!isestr(opt.pk)) {
        opt.pk = 'id'
    }
    if (!isbol(opt.autoGenPk)) {
        opt.autoGenPk = true
    }
    if (!isbol(opt.useStable)) {
        opt.useStable = true
    }
    if (!isestr(opt.storage)) {
        opt.storage = './worm.db' //storage, only use for sqlite
    }


    //儲存位址名稱(opt.db、opt.cl、opt.url、opt.storage)含孤立代理碼元者建構時即拋出(第66項, R13):
    //路徑與名稱經UTF-8轉換後孤立代理碼元被改寫為U+FFFD, 不同名稱會指向同一儲存(如兩個不同之storage路徑共用同一個sqlite檔);
    //須先於opt.url之解析(其錯誤訊息會回顯opt.url); 訊息為固定詞彙且不回顯值, 此處不檢NUL(名稱之NUL不屬本規則; sqlite之opt.storage之NUL另於dialect確定後檢, 見R20)
    for (let k of ['db', 'cl', 'url', 'storage']) {
        let why = chkOrmStr(opt[k], { nul: false })
        if (why !== null) {
            throw new Error(`invalid opt.${k}: ${why}`)
        }
    }

    let storageRaw = opt.storage //呼叫端所給之形式(相對路徑等), 供診斷輸出之秘密(儲存路徑)使用與sqlite之拒收判斷(R20); opt.storage於dialect確定後才path.resolve
    if (!isbol(opt.useEncryption)) {
        opt.useEncryption = false
    }


    //ee, 採wsemi之evem(即eventemitter3), 其於'error'無監聽者時僅回傳false而不拋出,
    //故本套件之操作行為不因呼叫端有無註冊監聽而改變; Node內建之EventEmitter具該拋出語義, 不可用
    let ee = evem()


    //opt.url解析失敗一律拋出, 因其屬呼叫端未履行契約, 且解析失敗後各函數皆無從運作,
    //建構為同步而無從以Promise.reject回報, 拋出即為reject之同步對應;
    //若僅console.log而回傳未綁定各函數之ee, 錯誤不經任何管道抵達呼叫端, 呼叫端僅得到
    //[w.select is not a function]之無關訊息, 與本規格[錯誤不得靜默]之通則相違;
    //訊息前綴與Error型態不變, 回顯之opt.url經wsemi之maskDiagUrl(R17-6, dspec Q2): 只回顯宣告之scheme(<scheme>://***), 不回顯主機、帳號與密碼(未捕獲時Node會把訊息印到stderr); maskDiagUrl為純函數, 不依賴尚未建立之diag
    let throwUrlErr = (msg) => {
        throw new Error(`${msg}, opt.url[${maskDiagUrl(opt.url, { schemes: DIALECTS })}]`)
    }


    //dialect
    let dialect
    ss = split(opt.url, '://')
    dialect = get(ss, 0, null)
    if (!dialect) {
        throwUrlErr('no dialect in opt.url')
    }
    if (!arrHas(dialect, DIALECTS)) {
        throwUrlErr('invalid dialect in opt.url')
    }


    //sqlite之opt.storage須為可指向實體檔案之路徑, 含NUL者與恰為':memory:'者建構時即拋出(第213項, R20; 僅sqlite, 其他方言不用opt.storage):
    //①含NUL: sqlite3於NUL截斷檔名(a\0b.sqlite開出名為a之檔), 不同名稱(a\0b、a\0c)會指向同一檔, 與孤立代理碼元(R13)同型之別名; 原因取wsemi之chkOrmStr(孤立代理已於上方迴圈檢出, 此處只會是NUL)
    //②':memory:'為sqlite之記憶體資料庫哨兵值, 但本套件各函數每次init+close, 記憶體庫無法跨呼叫保留, 且opt.storage經path.resolve後已非哨兵值(POSIX為cwd下名為:memory:之檔, win32為非法路徑), 故判斷用呼叫端所給之原值(storageRaw), 須於path.resolve之前
    //訊息為固定詞彙, 不回顯值(dspec輸入參數之檢測機制V20); 明確之檔案路徑寫法(如./:memory:)、形似者(:memory:x、:MEMORY:)為一般檔名, 不拒
    if (dialect === 'sqlite') {
        let whyStorage = chkOrmStr(storageRaw)
        if (whyStorage !== null) {
            throw new Error(`invalid opt.storage: ${whyStorage}`)
        }
        if (storageRaw === ':memory:') {
            throw new Error('invalid opt.storage: must be a file path, an in-memory database is not supported')
        }
    }
    opt.storage = path.resolve(opt.storage)


    u = get(ss, 1, '') //另存給後面使用


    //username, password
    let username
    let password
    ss = split(u, '@')
    u = get(ss, 1, '') //另存給後面使用
    ss = get(ss, 0, '')
    ss = split(ss, ':')
    if (size(ss) !== 2) {
        throwUrlErr('invalid username or password in opt.url')
    }
    username = get(ss, 0, '')
    password = get(ss, 1, '')
    if (dialect !== 'sqlite' || (dialect === 'sqlite' && opt.useEncryption)) {
        if (username === '') {
            throwUrlErr('invalid username in opt.url')
        }
        if (password === '') {
            throwUrlErr('invalid password in opt.url')
        }
    }


    //host, port
    let host
    let port
    ss = split(u, ':')
    host = get(ss, 0, null)
    port = get(ss, 1, null)
    if (dialect !== 'sqlite') {
        if (!host) {
            throwUrlErr('invalid host in opt.url')
        }
        if (!port) {
            throwUrlErr('invalid port in opt.url')
        }
    }


    //diag, 診斷輸出之單一出口(R17, wsemi之getDiag, 規則見wsemi之R-DIAG-01): src內所有console輸出與傳給sequelize之SQL紀錄函數皆經此, src不得直接呼叫console;
    //預設只輸出固定標籤與不含個資之摘要, opt.logDetail輸出遮罩後細節, opt.funLog取代console; 建構時取快照optLog(各選項型別錯誤者用預設), 其後各路徑共用, 不再讀取使用端的opt
    //cfg為本套件之宣告: 秘密(opt.url、帳號、密碼; sqlite另含資料庫檔案路徑)、樣式pragmaKey(sqlite加密時PRAGMA key之值)、代碼集合、錯誤類別名、巢狀錯誤鍵(sequelize包裝驅動錯誤之original、parent)、safe之鍵;
    //細節以detail清單遮罩(憑證、單獨帳號、儲存路徑), 對外之錯誤值(diag.err、diag.mask)以reject清單遮罩(只含憑證, dspec Q3)
    let optLog = mergeOptLog({ logging: false, logDetail: false, funLog: null }, opt)
    let cfg = {
        secrets: getDiagSecrets({ urls: [opt.url], username, password, paths: dialect === 'sqlite' ? [storageRaw, path.normalize(storageRaw), opt.storage] : [] }),
        patterns: ['pragmaKey'],
        codes: ['sqlite', 'mysql', 'mssql', 'sqlstate'],
        types: ERR_TYPES,
        nested: ['original', 'parent'],
        safeKeys: SAFE_KEYS,
    }
    let diag = getDiag(optLog, cfg)


    //logSql, 傳給sequelize之logging函數: sequelize以(sql全文, 查詢選項)呼叫, 查詢選項含綁定參數值故不取用, 只取第一個引數;
    //預設只輸出敘述種類, 完整sql(select等之where條件值為內嵌)為細節(logDetail時console.log('sql', sql全文)); logQueryParameters維持關閉, 故insert、update之綁定參數值($1…)不進sql紀錄
    let logSql = (sql) => {
        diag('sql', { stmt: getSqlStmt(sql) }, () => sql)
    }


    /**
     * 整批reject、error事件交出之錯誤值(R-DIAG, dspec D3)
     * 註: 本套件以固定詞彙自行產生者(ownErr登記之檢核與狀態錯誤、model載入失敗之訊息si.err)原樣交出: 固定字串經遮罩挖空即可反推短密碼
     * 註: 其餘(後端、驅動、model檔之hook、呼叫端數據之getter等外來值)經diag.err: 含憑證者為遮罩後之副本(instanceof同類、屬性保留), 無憑證者為原物件
     *
     * @ignore
     * @param {*} err 輸入攔得之錯誤值
     * @param {Object|null} si 輸入本次使用之實例(其err為model載入失敗之固定訊息), 無則null
     * @returns {*} 回傳交出之錯誤值
     */
    function outErr(err, si) {
        let own = isOwnErr(err)
        if (!own && si !== null) {
            try {
                own = err === si.err
            }
            catch (e) {
                //讀取實例拋錯者視為外來值
            }
        }
        return own ? err : diag.err(err)
    }


    /**
     * 取錯誤訊息字串，供逐筆結果之err欄位與error事件使用
     *
     * @ignore
     * @param {Error|String} err 輸入錯誤物件或字串
     * @returns {String} 回傳錯誤訊息字串
     */
    function getErrMsg(err) {

        //message
        let message = get(err, 'message')
        if (isestr(message)) {
            return message
        }

        return String(err)
    }


    /**
     * 發出change事件，訂閱函數拋錯不得影響本次操作之結果，故另包try並自行吞掉
     *
     * @ignore
     * @param {String} mode 輸入操作別字串
     * @param {Object|Array|null} data 輸入本次操作之數據
     * @param {Object|Array} res 輸入本次操作之結果
     * @returns {undefined} 無回傳值
     */
    function emitChange(mode, data, res) {
        try {
            ee.emit('change', mode, data, res)
        }
        catch (err) {
            diag('change handler threw', { mode, ...diag.brief(err) }, () => err)
        }
    }


    /**
     * 發出error事件，操作發生錯誤時發出，錯誤訊息一律轉為字串
     * 註: 事件僅為附加通知，所送出之資訊必另有正規管道——整批性錯誤經Promise.reject，逐筆失敗經該筆之err欄位
     * 註: 訂閱函數拋錯不得影響本次操作之結果，故另包try並自行吞掉
     * 註: 正常結果不得發出本事件，如insert全數已存在、save合併後內容相同、del主鍵未命中、selectByPk查無數據
     * 註: err須為已處置之值(R-DIAG): 外來值已經diag.err(整批之outErr)或diag.mask(逐筆err), 本套件之固定詞彙原樣, 與reject之值或逐筆err同源; data為呼叫端之輸入, 屬回傳契約不遮罩
     *
     * @ignore
     * @param {String} mode 輸入操作別字串
     * @param {Object|Array|null} data 輸入本次操作之數據
     * @param {Error|String} err 輸入已處置之錯誤物件或字串
     * @returns {undefined} 無回傳值
     */
    function emitError(mode, data, err) {
        try {
            ee.emit('error', mode, data, getErrMsg(err))
        }
        catch (errEmit) {
            diag('error handler threw', { mode, ...diag.brief(errEmit) }, () => errEmit)
        }
    }


    /**
     * 判定主鍵值是否有效(與主鍵欄型別無關之判定)
     * 註: select以外之五函數對主鍵之認定一律以本函數與normPkOf為準，令selectByPk之[命中]與insert、save、del內對既有數據之認定一致
     * 註: 有效即非空字串或有限數值(含0), new String與new Number取原始值; NaN、±Infinity、布林、物件等為無效(見cvFind.mjs之normPk)
     * 註: 含孤立代理碼元或NUL之字串亦為無效(第66項, R13)，故selectByPk回null；寫入側(insert、insertBulk、save、del)於procPk之前即由chkWf逐筆檢出，
     * 令autoGenPk為true時呼叫端所給之不合格字串不被當成[未給主鍵]而以自動產生之主鍵取代
     *
     * @ignore
     * @param {*} v 輸入主鍵值
     * @returns {Boolean} 回傳主鍵值是否有效布林值
     */
    function isEffPk(v) {
        return normPk(v, 'any').kind !== 'invalid'
    }


    /**
     * 判定是否為主鍵重複之錯誤
     * 註: sequelize於mssql、sqlite、mysql、mariadb、postgres皆將唯一鍵衝突轉為UniqueConstraintError，故此判定與dialect無關
     *
     * @ignore
     * @param {Error} err 輸入錯誤物件
     * @returns {Boolean} 回傳是否為主鍵重複錯誤布林值
     */
    function isDupPkError(err) {
        return err instanceof Sequelize.UniqueConstraintError
    }


    /**
     * 檢查呼叫端所給之實例是否已關閉，已關閉者拋出令各函數於入口快速失敗
     * 註: 實例已關閉為終態，後續操作必然失敗，屬影響全部數據之整批性錯誤，
     * 故不得降為逐筆ok為0而整批resolve，亦不得以正常空結果(null、未命中、空陣列)回應——
     * 後者使[已關閉]與[查無資料]同形，去重類呼叫端將把每筆判為未存在而靜默fail-open
     * 註: closeSequelize於關閉後將sequelize設為null，故據此判定即為精確訊號，
     * 不須倚賴驅動層之錯誤訊息比對(連線管理器關閉後所拋者為原生Error而非sequelize之錯誤類別)
     * 註: 內部自行初始化者(instance為null)每次呼叫各自開關連線，不受本檢查影響
     *
     * @ignore
     * @param {Object|null} instance 輸入呼叫端所給之實例物件
     * @returns {undefined} 無回傳值
     */
    function procClosed(instance) {

        //check
        if (instance === null) {
            return
        }

        //check
        if (sequelize === null) {
            throw ownErr(`instance is closed, need to use init() to get a new instance`)
        }

    }


    /**
     * 判定錯誤是否影響後續所有筆數而應視為整批性錯誤
     * 註: 依T4之判別原則，錯誤只影響該筆資料者為逐筆失敗，影響後續所有筆數者為整批性錯誤；
     * 連線層錯誤屬後者，故於逐筆函數內攔得時須往外拋而不得降為該筆ok為0
     * 註: 連線管理器關閉後所拋者為原生Error而非sequelize之ConnectionError，故另以訊息辨識
     *
     * @ignore
     * @param {Error} err 輸入錯誤物件
     * @returns {Boolean} 回傳是否為整批性錯誤布林值
     */
    function isBatchLevelError(err) {

        //check, sequelize之連線層錯誤家族
        if (err instanceof Sequelize.ConnectionError) {
            return true
        }

        //msg
        let msg = getErrMsg(err)

        //check
        if (msg.indexOf('connection manager was closed') >= 0) {
            return true
        }
        if (msg.indexOf('SQLITE_MISUSE') >= 0) {
            return true
        }

        return false
    }


    /**
     * 取cvFind所需之依賴物件
     * 註: attrs供欄名解析(嚴格鍵)與依欄位型別檢核條件之純量, pk令主鍵欄之條件值依normPk正規化而與主鍵函數一致
     *
     * @ignore
     * @param {Object} md 輸入資料表model物件
     * @returns {Object} 回傳{attrs, pk}
     */
    function getFindCtx(md) {
        return {
            attrs: get(md, 'rawAttributes', {}),
            pk: opt.pk,
        }
    }


    /**
     * 檢查model之預設scope是否會使select與delAll之對象分歧
     * 註: defaultScope帶limit或offset時, select依其order取前幾筆, 而sequelize之限量刪除不採order(sqlite為rowid IN子查詢),
     * 兩者命中之數據不同而會刪錯列, 故delAll遇此類model即拒絕
     * 註: defaultScope帶group或having時, findAll套用而destroy不套用(having過濾後只回部分列, 卻刪除全部符合where者);
     * 帶非空之include時, findAll以INNER JOIN篩列(required)而destroy不JOIN; 兩者亦拒絕
     * 註: defaultScope.where含Op.and者, delAll一律拒絕(維持既有處置): 原因為sequelize預設以Object.assign合併scope與cvFind輸出之where而被頂層Op.and覆寫,
     * delAll會刪到scope範圍外之列; 自R16起建構時設定whereMergeStrategy:'and', 兩者一律以AND結合(select亦正確套用該scope), 此拒絕為保守維持, 不再是越界之防線
     *
     * @ignore
     * @param {Object} md 輸入資料表model物件
     * @returns {undefined} 無回傳值
     */
    function chkScopeForDelAll(md) {
        let sc = get(md, '_scope', {})
        let bads = []
        for (let k of ['limit', 'offset']) {
            if (get(sc, k) !== undefined) {
                bads.push(k)
            }
        }
        for (let k of ['group', 'having']) {
            let v = get(sc, k)
            if (v !== undefined && v !== null && !(Array.isArray(v) && v.length === 0)) {
                bads.push(k)
            }
        }
        if (isearr(get(sc, 'include'))) {
            bads.push('include')
        }
        let w = get(sc, 'where')
        if (w !== null && typeof w === 'object' && Object.prototype.hasOwnProperty.call(w, Sequelize.Op.and)) {
            bads.push('where[Op.and]')
        }
        if (bads.length > 0) {
            throw ownErr(`model [${opt.cl}] has defaultScope with ${bads.join(', ')}, delAll can not delete the same rows as select`)
        }
    }


    /**
     * 正規化各筆數據之主鍵值並就地寫回, 型別不符者整批拋出
     * 註: 須於procPk之後、任何寫入之前一次完成; 已給主鍵而型別不符屬呼叫端未履行契約, 不以自動產生之主鍵取代
     * 註: 正規化值寫回數據, 令寫入值與其後各函數之比對值一致(如字串類主鍵欄之123存為'123'), 事件之數據亦為實際寫入者
     * 註: 訊息只含筆數索引與主鍵欄名, 不回顯主鍵值(第66項, R13)
     *
     * @ignore
     * @param {Object} md 輸入資料表model物件
     * @param {Array} data 輸入已補齊主鍵之數據陣列
     * @param {Array} [bads=[]] 輸入chkWf之逐筆訊息陣列, 非null者(save之不合格筆, 已另行處置為逐筆失敗)略過
     * @returns {undefined} 無回傳值
     */
    function normPks(md, data, bads = []) {
        data.forEach((v, k) => {
            if (get(bads, k, null) !== null) {
                return
            }
            let r = normPkOf(md, get(v, opt.pk))
            if (r.kind !== 'ok') {
                throw ownErr(`invalid data[${k}].${opt.pk}, type does not match pk column [${opt.pk}]`)
            }
            v[opt.pk] = r.v
        })
    }


    /**
     * 檢核各筆數據之DATE、DATEONLY欄值是否為年份在1–9999之有效時間(見cvFind.mjs之genDateChecker與inDateRange), 範圍外或無效者整批拋出
     * 註: 通過檢核之物件值(Date及其子類別、包裝物件、陣列等)由檢核函數就地換成已檢核之標準值(Date或YYYY-MM-DD文字)，令檢查之值即使用之值；操作對象為cloneDeep後之複本，呼叫端數據不變
     * 註: 須於normPks之後、任何寫入之前對全部筆數一次完成, 令拋錯時同批之有效筆數亦不會被寫入, 資料庫不變; 與條件側共用同一範圍判定
     * 註: 訊息只含筆數索引與欄名, 不回顯值; 同筆多欄範圍外者回報model屬性定義順序之第一個, 多筆範圍外者回報索引最小者
     * 註: sqlite以文字存DATE, 範圍外年份之文字與時間序不一致(哨兵new Date(8.64e15)存為275760-09-13…, 被$lt某日期誤中而遭delAll刪除), 故寫入時即拒收;
     * mssql之datetimeoffset、date範圍亦為0001–9999, 本函數於送出前即拒, 不依賴資料庫報錯, 兩方言行為一致
     *
     * @ignore
     * @param {Object} md 輸入資料表model物件
     * @param {Array} data 輸入已補齊並正規化主鍵之數據陣列
     * @param {Array} [bads=[]] 輸入chkWf之逐筆訊息陣列, 非null者(save之不合格筆, 已另行處置為逐筆失敗)略過
     * @returns {undefined} 無回傳值
     */
    function chkDates(md, data, bads = []) {
        let chk = genDateChecker(get(md, 'rawAttributes', {}))
        data.forEach((v, k) => {
            if (get(bads, k, null) !== null) {
                return
            }
            let col = chk(v)
            if (col !== null) {
                throw ownErr(`invalid data: date is not a valid time in years 1-9999 at [data[${k}].${col}]`)
            }
        })
    }


    /**
     * 依model之主鍵欄型別正規化主鍵值
     * 註: 規則見cvFind.mjs之normPk(與w-orm-mdb同一規則), 用於selectByPk、del、save、insert與insertBulk,
     * 並與select、delAll之主鍵欄條件共用, 令各函數對主鍵之命中判定一致; 字串類主鍵欄之有限數字以String(v)處置,
     * 整數類收安全整數與標準整數寫法之字串, DECIMAL之小數位不得多於scale, 型別不符者即規格之[主鍵值無效(型別不符)]而不送查詢
     *
     * @ignore
     * @param {Object} md 輸入資料表model物件
     * @param {*} v 輸入主鍵值
     * @returns {Object} 回傳{kind,v}, kind為'ok'、'invalid'或'mismatch'
     */
    function normPkOf(md, v) {
        let attrs = get(md, 'rawAttributes', {})
        return normPk(v, getColKind(attrs, opt.pk), getColScale(attrs, opt.pk))
    }


    /**
     * 檢查並補齊單筆數據之主鍵
     * autoGenPk為true時未帶有效主鍵者自動產生，為false時往外拋
     * 註: 未帶有效主鍵屬呼叫端未履行契約而非某一筆資料本身之問題，故為整批性錯誤而不降級為該筆ok為0，
     * 若降級為逐筆結果，呼叫端易於整批resolve之下漏看，使[忘了給主鍵]靜默變成[少寫了幾筆]
     * 註: 本函數須於任何寫入之前一次對全部數據完成，令拋錯時同批之有效筆數亦不會被寫入
     * 註: 自動產生之主鍵值為字串，故補值前檢查主鍵欄位確為字串類型，不符者以明確訊息拋出，
     * 而非任由資料庫回報型別錯誤; 本套件之model定義於手上得知主鍵型別，故不適用T6之[套件無從得知主鍵型別]例外
     *
     * @ignore
     * @param {Object} md 輸入資料表model物件
     * @param {Object} v 輸入數據物件
     * @param {Number} k 輸入數據於陣列內之索引
     * @returns {Object} 回傳補齊主鍵之數據物件
     */
    function procPk(md, v, k) {

        //check
        if (isEffPk(get(v, opt.pk))) {
            return v
        }

        //check, autoGenPk為false時主鍵須由呼叫端自備
        if (!opt.autoGenPk) {
            throw ownErr(`invalid data[${k}].${opt.pk}, autoGenPk is false`)
        }

        //at
        let at = get(md, `rawAttributes.${opt.pk}`)
        if (!iseobj(at)) {
            throw ownErr(`can not find pk column [${opt.pk}] in model [${opt.cl}]`)
        }

        //check, 自動產生之主鍵值為字串, 主鍵欄位須為字串類型方能容納
        let tp = get(at, 'type.key', '')
        if (!arrHas(tp, ['STRING', 'TEXT', 'CHAR', 'UUID', 'CITEXT'])) {
            throw ownErr(`autoGenPk is true but pk column [${opt.pk}] in model [${opt.cl}] is [${tp}], auto generated pk is a string, need to set autoGenPk to false and supply pk by caller`)
        }

        //genIDSeq, 為UUIDv7格式而具單調遞增性, 主鍵接近順序遞增可減少索引頁面分裂
        v[opt.pk] = genIDSeq()

        return v
    }


    /**
     * 初始化sequelize
     *
     * @memberOf WOrmReladb
     * @param {Boolean} [sync=false] 輸入當importModels匯入models時是否使用同步方式，將models資料變更至資料庫當中，此功能提供給createStorage之用
     * @returns {Promise} 回傳Promise，resolve代表關閉成功，reject回傳錯誤訊息
     */
    async function initSequelize(sync = false) {
        let err = null

        //optSeq
        let optSeq = {
            dialect,
            host,
            port,
            storage: opt.storage,
            logging: opt.logging ? logSql : false, //為函數時sequelize不再以console.log輸出(並免去logging:true之deprecation警告)
            logQueryParameters: false, //維持關閉(sequelize預設): 開啟時sql紀錄會附上綁定參數值
            define: {
                //underscored: false,
                //freezeTableName: false,
                //syncOnAssociation: true,
                //charset: 'utf8',
                //dialectOptions: {
                //  collate: 'utf8_general_ci'
                //},
                timestamps: false,
                //model之defaultScope.where與select、delAll等傳入之where一律以AND結合(R16): sequelize預設為overwrite, 同名頂層鍵後者覆寫,
                //scope.where為Sequelize.literal或Sequelize.where(非一般物件)時scope直接被丟棄, 致select與delAll越出scope
                whereMergeStrategy: 'and'
            },
            // pool: {
            //     max: 20,
            //     min: 0,
            //     acquire: 30000, //The maximum time, in milliseconds, that pool will try to get connection before throwing error
            //     idle: 10000 //The maximum time, in milliseconds, that a connection can be idle before being released.
            // },
            //sync: { force: true }, //強制同步
        }
        if (opt.useEncryption && dialect === 'sqlite') {
            optSeq.dialectModulePath = '@journeyapps/sqlcipher' //sqlite加密版
        }

        //sequelize, 建構(缺少驅動等同步拋出)與sqlite加密設定之例外為外來值(加密設定失敗之錯誤含PRAGMA key敘述, 即帳號:密碼), 經diag.err後拋出(R-DIAG);
        //init之reject與各函數之整批reject皆源於此, 由各函數之outErr再經diag.err時已無憑證而回原物件
        try {
            sequelize = new Sequelize(opt.db, username, password, optSeq)
            if (opt.useEncryption && dialect === 'sqlite') {
                await sequelize.query('PRAGMA cipher_compatibility = 4') //設置sqlite密鑰相容性
                await sequelize.query(`PRAGMA key = '${username}:${password}'`) //設置slqite密碼為${username}:${password}, 為使用者名稱與密碼用冒號分隔
            }
        }
        catch (errSeq) {
            throw diag.err(errSeq)
        }

        //importModels, 若model內id不是pk則需要強制更改成為pk, 否則sequelize無法匯入; 其錯誤只進診斷輸出之細節, 對外為本套件之固定訊息err
        let r = await pm2resolve(importModels)(opt.fdModels, sequelize, opt.cl, { sync, type: opt.modelType })

        //check
        if (r.state === 'error') {
            diag('can not import model', { cl: opt.cl, ...diag.brief(r.msg) }, () => r)
            err = `can not import model: ${opt.cl}, need to use genModelsByDB, genModelsByTabs or create ${opt.cl}.${opt.modelType}`
        }

        //mds
        let mds = r.msg

        //model自身指定whereMergeStrategy時以model選項為準而蓋過上列define預設(scope又被覆寫或丟棄), 故匯入後一律設為'and'(R16)
        if (err === null) {
            for (let md of Object.values(mds)) {
                md.options.whereMergeStrategy = 'and'
            }
        }

        return {
            mds,
            err,
            close: () => {
                return closeSequelize('external')
            },
        }
    }


    /**
     * 關閉sequelize
     *
     * 註: sqlite之storage無法開啟(資料夾、檔名超長、無權限、路徑非法等SQLITE_CANTOPEN)時，sequelize仍把開啟失敗之Database留在連線表，其close()永不回呼而使sequelize.close()永不settle(各函數隨之永不settle)；
     * 故sqlite於close前先移除未開啟之Database(dropUnopenedConn，R19)，使開啟失敗之錯誤得以既有通道交出
     *
     * @memberOf WOrmReladb
     * @returns {Promise} 回傳Promise，resolve代表關閉成功，reject回傳錯誤訊息
     */
    async function closeSequelize(from) {
        if (sequelize !== null) {
            try {
                if (dialect === 'sqlite') {
                    dropUnopenedConn(sequelize)
                }
                await sequelize.close()
            }
            catch (err) {
                throw diag.err(err) //驅動之關閉錯誤為外來值(R-DIAG); 各函數之finally與init回傳之close皆經此
            }
            //console.log(from, 'sequelize.close()')
        }
        sequelize = null
        //console.log(from, 'sequelize = null')
    }

        }
        else {

    /**
     * 查詢數據
     *
     * 使用須知(條件find為系列白名單, 規則同delAll, 見CLAUDE_rulebook.md之R11):
     * 1. 不給條件即查詢全部數據: find未給、為undefined或為{}時回傳全部數據; 給了條件才是部分查詢。
     * 2. 傳了值卻不是條件物件(null、陣列含[]與[{...}]、字串、數值、布林、函數、Date、RegExp、Map、Set、類別實例等)
     *    一律整批reject, 不視為{}。
     * 3. 條件僅支援系列白名單: 欄位等值{欄:純量}(多欄為AND)、欄位運算子$eq $ne $gt $gte $lt $lte(值為純量)與
     *    $in $nin(值為純量陣列)、文件層$and $or(值為條件物件之非空陣列, 最多999項)。純量為字串、有限數字、布林、有效Date;
     *    null僅可用於欄位等值、$eq、$ne。白名單外之運算子($not $nor $regex $options $exists $where $expr $like $ilike等)、
     *    sequelize之Op Symbol、RegExp值、陣列值(請改用$in)、欄位層$and/$or、函數值、Symbol鍵、undefined值、空的$nin/$and/$or等
     *    一律整批reject, 訊息為invalid find: 原因 at [路徑], 且於載入model與連線前即檢核。
     * 4. 欄名須完全等於model之屬性名或資料庫欄名(區分大小寫), 且為字母或底線開頭之字母/數字/底線(最多128字元);
     *    純量須符合欄位型別(如數值欄給'abc'會reject); 主鍵欄之條件值依主鍵欄型別正規化(字串類主鍵欄之{id:123}即{id:'123'})。
     * 5. 與delAll共用同一檢核與同一條件, select(find)命中之數據即delAll(find)刪除之數據。
     * 6. 語意差異(SQL式): $ne/$nin不命中欄位為NULL之數據(三值邏輯)。需明確表達時可用可攜寫法:
     *    排除NULL為{$and:[{f:{$ne:null}},{f:{$ne:x}}]}, 含NULL為{$or:[{f:null},{f:{$ne:x}}]}; 字串比對之大小寫依資料庫定序。
     *
     * 註: 恆回傳陣列, 無符合數據回傳空陣列
     *
     * @memberOf WOrmReladb
     * @param {Object} [find={}] 輸入查詢條件物件(系列白名單, 規則同delAll之使用須知第1-4點); 未給、undefined或{}回傳全部數據; 非條件物件或白名單外運算子整批reject
     * @param {Object} [option={}] 輸入設定物件，預設為{}
     * @param {Object} [option.instance=null] 輸入實例instance物件，預設為null
     * @param {Object} [option.transaction=null] 輸入交易(transaction)物件，預設為null
     * @returns {Promise} 回傳Promise，resolve回傳數據陣列，reject回傳錯誤訊息
     */
    async function select(find = {}, option = {}) {

        //instance
        let instance = get(option, 'instance', null)

        //transaction
        let transaction = get(option, 'transaction', null)

        //check
        if (opt.useEncryption && dialect === 'sqlite') {
            if (transaction !== null) {
                diag('@journeyapps/sqlcipher can not support transaction.')
            }
        }

        //isErr, res
        let isErr = false
        let res = null

        //si
        let si = instance

        //inited, 本次呼叫是否已進入initSequelize(僅此時finally才關閉sequelize; 預檢失敗於其前者不得關閉呼叫端先前持有之連線)
        let inited = false

        try {

            //check, 實例已關閉屬整批性錯誤, 於入口快速失敗
            procClosed(instance)

            //chkFindShape, 條件形狀不合法者(系列白名單, R11)於載入model與建立sequelize實例之前即拋錯: 不連線、不送出任何SQL, reject並發出error事件; 其後只使用檢核後之複本
            let fd = chkFindShape(find)

            //initSequelize, 內部自動初始化須在try內: 其例外(如驅動缺失)與si.err同屬整批性錯誤, 方能於reject前發出error事件, 已建構之sequelize亦由finally關閉
            if (instance === null) {
                inited = true
                si = await initSequelize()
            }

            //check, 連線或匯入models失敗影響本次操作之全部數據, 屬整批性錯誤
            if (si.err) {
                throw si.err
            }

            //md
            let md = si.mds[opt.cl]

            //useFind, 與delAll共用cvFind(對複本再檢核一次, 冪等), 欄名、欄位型別等翻譯層不合法者於此拋錯而reject並發出error事件
            let useFind = cvFind(fd, getFindCtx(md))

            //setting, raw令回傳純數據物件而不含資料庫內部欄位
            let setting = {
                where: useFind,
                raw: true,
            }
            if (transaction !== null) {
                setting.transaction = transaction
            }

            //findAll, 恆回傳陣列, 無符合數據為空陣列
            res = await md.findAll(setting)
            if (!isarr(res)) {
                res = []
            }

        }
        catch (err) {
            isErr = true
            res = outErr(err, si) //外來值經diag.err, 本套件之固定詞彙原樣; reject之值與error事件同源
        }
        finally {

            //closeSequelize
            if (instance === null && inited) { //內部自動初始化得close
                await closeSequelize('select')
            }

        }

        //check
        if (isErr) {
            emitError('select', null, res)
            return Promise.reject(res)
        }

        return res
    }


    /**
     * 由主鍵查詢單筆數據，因由資料庫依主鍵索引取值且僅回傳單筆，不需如select提取全部符合數據再處理，故數據量大時效能較佳
     *
     * 註: 主鍵欄位取自opt.pk，預設為id，支援由呼叫端指定
     * 註: 主鍵未命中或主鍵值無效皆回傳null而不reject，[命中]之判定基準與insert、save、del內對既有數據之認定一致
     * 註: 含孤立代理碼元或NUL之字串主鍵(含new String包裝者)為無效而回傳null，不連線、不送查詢(第66項, R13): UTF-8存放之資料庫會把孤立代理碼元改寫為U+FFFD，若照送會命中他筆
     * 註: 主鍵值依主鍵欄型別正規化後查詢: 字串類主鍵欄之有限數字以String(v)處置(selectByPk(123)即查'123'), 整數類主鍵欄之標準整數寫法字串轉為數值;
     * 型別不符(整數類主鍵欄給'012'、12.4、'abc'等)亦屬主鍵值無效而回傳null
     * 註: 本函數不得有副作用，故不創建資料表
     *
     * @memberOf WOrmReladb
     * @param {String|Number} pk 輸入主鍵值，即數據內opt.pk欄位之值
     * @param {Object} [option={}] 輸入設定物件，預設為{}
     * @param {Object} [option.instance=null] 輸入實例instance物件，預設為null
     * @param {Object} [option.transaction=null] 輸入交易(transaction)物件，預設為null
     * @returns {Promise} 回傳Promise，resolve回傳數據物件，若無此主鍵或主鍵值無效則回傳null，reject回傳錯誤訊息
     */
    async function selectByPk(pk, option = {}) {

        //instance
        let instance = get(option, 'instance', null)

        //transaction
        let transaction = get(option, 'transaction', null)

        //check, 實例已關閉之判定須先於[主鍵值無效回null], 否則兩者同形而使[已關閉]被誤讀為[查無資料]
        try {
            procClosed(instance)
        }
        catch (err) {
            emitError('selectByPk', null, err)
            return Promise.reject(err)
        }

        //check, 未給有效主鍵值視為查無數據, 不送查詢以免無效值被轉為null而誤中其他數據;
        //含孤立代理碼元或NUL之字串亦為無效(R13); 包裝物件(new String、new Number)只求值一次, 其後以原始值查詢(檢查之值即使用之值)
        let r0 = normPk(pk, 'any')
        if (r0.kind !== 'ok') {
            return null
        }
        pk = r0.v

        //check
        if (opt.useEncryption && dialect === 'sqlite') {
            if (transaction !== null) {
                diag('@journeyapps/sqlcipher can not support transaction.')
            }
        }

        //isErr, res
        let isErr = false
        let res = null

        //si
        let si = instance

        try {

            //check, 實例已關閉屬整批性錯誤, 於入口快速失敗
            procClosed(instance)

            //initSequelize, 內部自動初始化須在try內: 其例外(如驅動缺失)與si.err同屬整批性錯誤, 方能於reject前發出error事件, 已建構之sequelize亦由finally關閉
            if (instance === null) {
                si = await initSequelize()
            }

            //check
            if (si.err) {
                throw si.err
            }

            //md
            let md = si.mds[opt.cl]

            //normPkOf, 依主鍵欄型別正規化, 型別不符即主鍵值無效而回null且不送查詢(送出時兩方言皆會把整數欄之'012'轉為12而命中)
            let r = normPkOf(md, pk)
            if (r.kind !== 'ok') {
                res = null
            }
            else {

                //setting
                let setting = {
                    where: { [opt.pk]: r.v },
                    raw: true,
                }
                if (transaction !== null) {
                    setting.transaction = transaction
                }

                //findOne
                let v = await md.findOne(setting)

                //check, 判定基準與insert、save、del內對既有數據之認定一致
                if (iseobj(v)) {
                    res = v
                }
                else {
                    //不存在主鍵, 回傳null
                    res = null
                }

            }

        }
        catch (err) {
            isErr = true
            res = outErr(err, si) //外來值經diag.err, 本套件之固定詞彙原樣; reject之值與error事件同源
        }
        finally {

            //closeSequelize
            if (instance === null) { //內部自動初始化得close
                await closeSequelize('selectByPk')
            }

        }

        //check
        if (isErr) {
            emitError('selectByPk', null, res)
            return Promise.reject(res)
        }

        return res
    }


    /**
     * 插入數據，僅於主鍵不存在時寫入，已存在者跳過且不覆寫
     *
     * 由資料庫於主鍵之唯一約束上原子完成[檢查主鍵不存在]與[寫入]，併發時同一主鍵僅有一次成功
     *
     * 註: n為輸入筆數即本次嘗試插入之基準，nInserted為實際插入筆數，全數已存在而nInserted為0屬正常結果
     * 註: 同批含重複主鍵時僅首筆計入nInserted，其餘視為已存在
     * 註: opt.autoGenPk為true(預設)時未帶有效主鍵者自動產生，為false時未帶有效主鍵即reject且同批皆不寫入
     * 註: 主鍵值依主鍵欄型別正規化後寫入(字串類主鍵欄之123存為'123')；型別不符(整數類主鍵欄給'012'等)即reject且同批皆不寫入，不以自動產生之主鍵取代
     * 註: DATE、DATEONLY欄之值(Date、epoch數值、日期字串)換算後年份須在1–9999(DATE依UTC，DATEONLY依存入之日期文字)，範圍外者(如new Date(8.64e15)、'+010000-01-01T00:00:00Z')即reject且同批皆不寫入，
     * 錯誤訊息為invalid data: date is not a valid time in years 1-9999 at [data[索引].欄名]，不回顯值；sqlite以文字存日期，範圍外年份之文字與時間序不一致，會使以日期為條件之delAll刪錯列，
     * 故兩方言一律於送出前拒收；無效時間(Invalid Date、NaN、無法解析之字串、空字串等，sqlite會存為字典序大於任何年份之文字Invalid date，mssql則由資料庫報錯)亦一併reject；
     * 庫內既有之範圍外日期不遷移，須自行清理或改寫(見README)
     * 註: 主鍵與數據之字串須良構(第66項, R13): 主鍵不得含孤立代理碼元(UTF-16代理對被拆開後只剩一半, UTF-8存放時被改寫為U+FFFD而使不同主鍵寫成同一鍵)或NUL，
     * 數據之字串值與鍵名(含巢狀物件、陣列元素、Map/Set等，不論欄位是否在model內)不得含孤立代理碼元(值之NUL不拒)；U+FFFD本身與成對代理(emoji等)為良構，照常接受。
     * 任一筆不合格即reject且同批皆不寫入，訊息為invalid id: 原因 at [data[索引].主鍵欄]或invalid data: 原因 at [data[索引].路徑]，不回顯值；
     * 不隨opt.autoGenPk改變(autoGenPk為true時亦不以自動產生之主鍵取代呼叫端所給之不合格主鍵)，檢核先於連線與model載入，作用於複本
     * 註: option.returnList開啟時改回與輸入等長且保序之逐筆結果陣列[{n,nInserted,ok}]，
     * 聚合計數只答得出[有幾筆是新的]而答不出[是哪幾筆]，而後者正是去重之產出物；
     * 逐筆元素之n與ok恆為1，資訊由nInserted承載，不變式為
     * 陣列長度等於輸入筆數且filter(v => v.nInserted === 1).length等於聚合模式之nInserted
     *
     * @memberOf WOrmReladb
     * @param {Object|Array} data 輸入數據物件或陣列
     * @param {Object} [option={}] 輸入設定物件，預設為{}
     * @param {Object} [option.instance=null] 輸入實例instance物件，預設為null
     * @param {Object} [option.transaction=null] 輸入交易(transaction)物件，預設為null
     * @param {Boolean} [option.returnList=false] 輸入是否改回逐筆結果陣列布林值，預設false
     * @returns {Promise} 回傳Promise，resolve回傳插入結果{n,nInserted,ok}，或於returnList開啟時回傳[{n,nInserted,ok}]，reject回傳錯誤訊息
     */
    async function insert(data, option = {}) {

        //returnList, 回傳形式之切換為靜態, 僅由本選項之取值決定, 不因數據內容或執行結果而變
        let returnList = get(option, 'returnList', false) === true

        //check
        if (!iseobj(data) && !isearr(data)) {
            if (returnList) {
                return []
            }
            return {
                n: 0,
                nInserted: 0,
                ok: 1,
            }
        }

        //instance
        let instance = get(option, 'instance', null)

        //transaction
        let transaction = get(option, 'transaction', null)

        //check
        if (opt.useEncryption && dialect === 'sqlite') {
            if (transaction !== null) {
                diag('@journeyapps/sqlcipher can not support transaction.')
            }
        }

        //isErr, res
        let isErr = false
        let res = null

        //si
        let si = instance

        //inited, 本次呼叫是否已進入initSequelize(僅此時finally才關閉sequelize; 預檢失敗於其前者不得關閉呼叫端先前持有之連線)
        let inited = false

        try {

            //check, 實例已關閉屬整批性錯誤, 於入口快速失敗
            procClosed(instance)

            //chkWf, 主鍵與資料之字串良構(孤立代理碼元、主鍵之NUL)逐筆檢核(R13): 先於連線、model載入與任何寫入, 檢查之物件即cloneDeep後之複本且其後只使用該複本;
            //任一筆不合格即整批拋出(以索引最小者為準)而不寫入任何一筆, 不隨autoGenPk改變, 不以自動產生之主鍵取代不合格之主鍵
            let bads = chkWf(isarr(data) ? data : [data], opt.pk, true)
            let iBad = bads.findIndex((v) => v !== null)
            if (iBad >= 0) {
                throw ownErr(bads[iBad])
            }

            //initSequelize, 內部自動初始化須在try內: 其例外(如驅動缺失)與si.err同屬整批性錯誤, 方能於reject前發出error事件, 已建構之sequelize亦由finally關閉
            if (instance === null) {
                inited = true
                si = await initSequelize()
            }

            //check
            if (si.err) {
                throw si.err
            }

            //md
            let md = si.mds[opt.cl]

            //check
            if (!isarr(data)) {
                data = [data]
            }

            //procPk, 須於任何寫入之前一次完成, 令autoGenPk為false而拋錯時同批之有效筆數亦不會被寫入
            data = map(data, (v, k) => {
                return procPk(md, v, k)
            })

            //normPks, 主鍵值依主鍵欄型別正規化後寫回, 型別不符者整批reject且不寫入任何一筆, 令insert對既有數據之認定與selectByPk一致
            normPks(md, data)

            //chkDates, DATE、DATEONLY欄之值須為年份在1–9999之有效時間, 範圍外或無效者整批reject且不寫入任何一筆(與條件側共用範圍判定, 見R12)
            chkDates(md, data)

            //nAll, n之基準為輸入筆數
            let nAll = size(data)

            //setting
            let setting = { }
            if (transaction !== null) {
                setting.transaction = transaction
            }

            //create, 逐筆插入以取得精確之nInserted
            //註: 不採bulkCreate配合ignoreDuplicates, 因mssql未宣告該能力而選項會被靜默忽略仍撞唯一約束,
            //且bulkCreate無從回報實際插入筆數; 單筆INSERT本即由主鍵之唯一約束原子完成檢查與寫入
            //註: bIns即逐筆判定, 與輸入等長且保序, 供returnList直接包裝而不須另行推導
            let bInss = await pmSeries(data, async(v) => {

                //bIns, 撞既有主鍵者視為已存在而跳過, 其餘錯誤影響全部數據須往外拋
                let bIns = await md.create(v, setting)
                    .then(() => {
                        return true
                    })
                    .catch((err) => {
                        if (isDupPkError(err)) {
                            return false
                        }
                        throw err
                    })

                return bIns
            })

            if (returnList) {

                //res, 與輸入等長且保序之逐筆結果; 各筆之n恆為1(該筆主鍵或為命中既有或經插入而產生),
                //ok恆為1(insert之任何錯誤皆屬整批性錯誤而reject), 資訊由nInserted承載
                res = map(bInss, (bIns) => {
                    return {
                        n: 1,
                        nInserted: bIns ? 1 : 0,
                        ok: 1,
                    }
                })

            }
            else {

                //res, 全數已存在而nInserted為0屬正常結果, 不視為錯誤
                res = {
                    n: nAll,
                    nInserted: size(bInss.filter((v) => v === true)),
                    ok: 1,
                }

            }

        }
        catch (err) {
            isErr = true
            res = outErr(err, si) //外來值經diag.err, 本套件之固定詞彙原樣; reject之值與error事件同源
        }
        finally {

            //closeSequelize
            if (instance === null && inited) { //內部自動初始化得close
                await closeSequelize('insert')
            }

        }

        //check
        if (isErr) {
            emitError('insert', data, res)
            return Promise.reject(res)
        }

        //emit, 須於結果定案後發出
        emitChange('insert', data, res)

        return res
    }


    /**
     * 批次插入數據，全批視為一個單位：全部插入成功，或一筆都不寫入
     *
     * 本函數非insert之加速版，兩者衝突政策不同：insert於主鍵已存在時跳過該筆且整批ok為1，
     * 本函數則整批reject且不寫入任何一筆，同批含重複主鍵者亦視為衝突。
     * 確無衝突時兩者之可觀察結果完全相同，差異僅於有衝突時顯現。
     *
     * 註: n為輸入筆數即本次嘗試插入之基準；nInserted於成功時恆等於n，雖無額外資訊仍保留，
     * 令呼叫端得與insert共用同一段結果處理程式碼
     * 註: 不提供逐筆結果，故不出現ok為0與err；需要逐筆處置者改用insert
     * 註: opt.autoGenPk為true(預設)時未帶有效主鍵者自動產生，為false時未帶有效主鍵即reject且同批皆不寫入
     * 註: 主鍵值依主鍵欄型別正規化後寫入(字串類主鍵欄之123存為'123')；型別不符(整數類主鍵欄給'012'等)即reject且同批皆不寫入
     * 註: DATE、DATEONLY欄之值換算後年份須在1–9999，範圍外或無效者(如new Date(8.64e15)、new Date(NaN))即reject且同批皆不寫入，訊息為invalid data: date is not a valid time in years 1-9999 at [data[索引].欄名]，
     * 不回顯值，兩方言一致於送出前拒收，規則同insert
     * 註: 主鍵與數據之字串須良構(孤立代理碼元、主鍵之NUL)，不合格者整批reject且同批皆不寫入，規則同insert(第66項, R13)
     * 註: [全有全無]為獨立於T7之額外要求——T7僅保證每筆之[檢查主鍵不存在]與[寫入]為原子，不保證整批。
     * mssql因綁定參數數量上限會由驅動層自動拆為多語句送出，實測1000筆之批次於末筆撞主鍵時已有946筆落盤，
     * 故一律以交易包覆並於失敗時回滾；呼叫端已給transaction時改以巢狀交易(SAVEPOINT)包覆，
     * 令回滾範圍限於本次呼叫而不影響呼叫端交易內先前之寫入
     *
     * @memberOf WOrmReladb
     * @param {Object|Array} data 輸入數據物件或陣列
     * @param {Object} [option={}] 輸入設定物件，預設為{}
     * @param {Object} [option.instance=null] 輸入實例instance物件，預設為null
     * @param {Object} [option.transaction=null] 輸入交易(transaction)物件，預設為null
     * @returns {Promise} 回傳Promise，resolve回傳插入結果{n,nInserted,ok}，reject回傳錯誤訊息
     */
    async function insertBulk(data, option = {}) {

        //check
        if (!iseobj(data) && !isearr(data)) {
            return {
                n: 0,
                nInserted: 0,
                ok: 1,
            }
        }

        //instance
        let instance = get(option, 'instance', null)

        //transaction
        let transaction = get(option, 'transaction', null)

        //check
        if (opt.useEncryption && dialect === 'sqlite') {
            if (transaction !== null) {
                diag('@journeyapps/sqlcipher can not support transaction.')
            }
        }

        //isErr, res
        let isErr = false
        let res = null

        //si
        let si = instance

        //inited, 本次呼叫是否已進入initSequelize(僅此時finally才關閉sequelize; 預檢失敗於其前者不得關閉呼叫端先前持有之連線)
        let inited = false

        try {

            //check, 實例已關閉屬整批性錯誤, 於入口快速失敗
            procClosed(instance)

            //chkWf, 主鍵與資料之字串良構(孤立代理碼元、主鍵之NUL)逐筆檢核(R13): 先於連線、model載入與任何寫入, 檢查之物件即cloneDeep後之複本且其後只使用該複本;
            //任一筆不合格即整批拋出(以索引最小者為準)而不寫入任何一筆, 不隨autoGenPk改變, 不以自動產生之主鍵取代不合格之主鍵
            let bads = chkWf(isarr(data) ? data : [data], opt.pk, true)
            let iBad = bads.findIndex((v) => v !== null)
            if (iBad >= 0) {
                throw ownErr(bads[iBad])
            }

            //initSequelize, 內部自動初始化須在try內: 其例外(如驅動缺失)與si.err同屬整批性錯誤, 方能於reject前發出error事件, 已建構之sequelize亦由finally關閉
            if (instance === null) {
                inited = true
                si = await initSequelize()
            }

            //check
            if (si.err) {
                throw si.err
            }

            //md
            let md = si.mds[opt.cl]

            //check
            if (!isarr(data)) {
                data = [data]
            }

            //procPk, 須於任何寫入之前一次完成, 令autoGenPk為false而拋錯時同批皆不會被寫入
            data = map(data, (v, k) => {
                return procPk(md, v, k)
            })

            //normPks, 主鍵值依主鍵欄型別正規化後寫回, 型別不符者整批reject且不寫入任何一筆
            normPks(md, data)

            //chkDates, DATE、DATEONLY欄之值須為年份在1–9999之有效時間, 範圍外或無效者整批reject且不寫入任何一筆(與條件側共用範圍判定, 見R12)
            chkDates(md, data)

            //nAll, n之基準為輸入筆數
            let nAll = size(data)

            //useTx, sqlcipher不支援transaction, 該情形下改倚賴sqlite單一語句之原子性
            let useTx = !(opt.useEncryption && dialect === 'sqlite')

            //t, 呼叫端未給交易時自開; 已給時以巢狀交易(SAVEPOINT)包覆令回滾範圍限於本次呼叫
            let t = null
            if (useTx) {
                t = (transaction !== null)
                    ? await md.sequelize.transaction({ transaction })
                    : await md.sequelize.transaction()
            }

            //setting
            let setting = {}
            if (t !== null) {
                setting.transaction = t
            }
            else if (transaction !== null) {
                setting.transaction = transaction
            }

            try {

                //bulkCreate, 未加任何跳過選項, 故任一筆撞主鍵即整句失敗, 同批重複主鍵亦於此被偵測為衝突
                await md.bulkCreate(data, setting)

                if (t !== null) {
                    await t.commit()
                }

            }
            catch (err) {
                if (t !== null) {
                    await t.rollback().catch(() => {})
                }
                throw err
            }

            //res, 全有全無故成功時nInserted恆等於n
            res = {
                n: nAll,
                nInserted: nAll,
                ok: 1,
            }

        }
        catch (err) {
            isErr = true
            res = outErr(err, si) //外來值經diag.err, 本套件之固定詞彙原樣; reject之值與error事件同源
        }
        finally {

            //closeSequelize
            if (instance === null && inited) { //內部自動初始化得close
                await closeSequelize('insertBulk')
            }

        }

        //check
        if (isErr) {
            emitError('insertBulk', data, res)
            return Promise.reject(res)
        }

        //emit, 須於結果定案後發出
        emitChange('insertBulk', data, res)

        return res
    }


    /**
     * 儲存單筆數據
     *
     * 採條件寫入配合衝突偵測與重試: 每次寫入本身為單一條件式原子語句——插入由主鍵之唯一約束判定、
     * 更新由WHERE主鍵之比對判定，皆非由預讀值決定其成敗；預讀僅用於選擇發出哪一條語句，
     * 選錯不會產生錯誤結果，只會被偵測並重試(插入撞既有主鍵、更新未命中任何列)，故縱使預讀值已過期，
     * 最終結果仍與該次操作單獨執行時相同。
     * 註: 不採sequelize之upsert，因sqlite無從於單一語句內回報本次係插入或更新(其created恆為null)，
     * 而insert與save之規格要求nInserted精確，兩者於sqlite互斥，故改採本形式
     * 註: [內容相同]之判定基準為將待寫入物件合併進現值後與現值相同，相同則不寫入而nModified為0，
     * 合併取淺層以與後端整欄取代之寫入行為一致，令nModified忠實反映資料庫端是否真的寫入
     *
     * @ignore
     * @param {Object} md 輸入資料表model物件
     * @param {Object} v 輸入數據物件
     * @param {Boolean} autoInsert 輸入是否於查無數據時自動改以插入處理布林值
     * @param {Object|null} transaction 輸入交易(transaction)物件
     * @param {Array} ks 輸入model之欄位名陣列
     * @param {Number} k 輸入數據於陣列內之索引，供訊息使用(訊息不回顯主鍵值)
     * @returns {Promise} 回傳Promise，resolve回傳本筆儲存結果，本筆失敗時亦resolve並以ok為0回報
     */
    async function saveOne(md, v, autoInsert, transaction, ks, k) {

        //nTry, 預測錯誤時重試, 給予有限次數令其收斂
        let nTry = 3

        //rest
        let rest = null

        //inserted, 供結果定案後發出insert事件用
        let inserted = false

        //pkv
        let pkv = get(v, opt.pk)

        //setting
        let settingRead = {
            where: { [opt.pk]: pkv },
            raw: true,
        }
        let settingUpd = {
            where: { [opt.pk]: pkv },
        }
        let settingIns = {}
        if (transaction !== null) {
            settingRead.transaction = transaction
            settingUpd.transaction = transaction
            settingIns.transaction = transaction
        }

        for (let i = 0; i < nTry; i++) {

            try {

                //預讀, 供判斷內容是否相同之快速路徑, 其結果不用於決定寫入內容
                let _v = await md.findOne(settingRead)

                if (iseobj(_v)) {
                    //主鍵存在

                    //_vt, vt, 以待寫入物件之非主鍵欄位淺層覆蓋現值後與現值比對
                    let _vt = omit(_v, opt.pk)
                    let vt = {
                        ..._vt,
                        ...pick(omit(v, opt.pk), ks),
                    }

                    //check, 內容相同者不寫入
                    if (isEqual(vt, _vt)) {
                        rest = {
                            n: 1,
                            nInserted: 0,
                            nModified: 0,
                            ok: 1,
                        }
                        break
                    }

                    //update, 單一UPDATE語句由WHERE主鍵原子完成查找與更新
                    let rr = await md.update(v, settingUpd)
                    let nAff = get(rr, 0, 0)

                    if (nAff > 0) {
                        rest = {
                            n: 1,
                            nInserted: 0,
                            nModified: 1,
                            ok: 1,
                        }
                        break
                    }

                    //該列已被他方刪除, 重試改走插入
                    continue
                }

                //主鍵不存在且未開啟autoInsert則不寫入以免無中生有
                if (!autoInsert) {
                    rest = {
                        n: 0,
                        nInserted: 0,
                        nModified: 0,
                        ok: 1,
                    }
                    break
                }

                //create, 單一INSERT語句由主鍵之唯一約束原子完成檢查與寫入
                let bIns = await md.create(v, settingIns)
                    .then(() => {
                        return true
                    })
                    .catch((err) => {
                        if (isDupPkError(err)) {
                            return false
                        }
                        throw err
                    })

                if (bIns) {
                    rest = {
                        n: 1,
                        nInserted: 1,
                        nModified: 0,
                        ok: 1,
                    }
                    inserted = true
                    break
                }

                //他方已插入同一主鍵, 重試改走更新
                continue
            }
            catch (err) {

                //check, 連線層錯誤影響後續所有筆數, 依T4屬整批性錯誤, 往外拋而不降為本筆ok為0
                if (isBatchLevelError(err)) {
                    throw err
                }

                //其餘錯誤視為本筆失敗, 不中斷整批, 由呼叫端以ok與err判讀; err為後端之外來值, 建立時即經diag.mask(R-DIAG), 令回傳值與error事件同源
                rest = {
                    n: 1,
                    nInserted: 0,
                    nModified: 0,
                    ok: 0,
                    err: diag.mask(getErrMsg(err)),
                }
                break
            }

        }

        //check, 重試耗盡仍未定案
        if (rest === null) {
            rest = {
                n: 1,
                nInserted: 0,
                nModified: 0,
                ok: 0,
                err: `can not save data[${k}].${opt.pk} in ${nTry} tries`,
            }
        }

        //emit, 須於本筆結果定案後發出, 避免訂閱函數拋錯影響本筆結果
        if (inserted) {
            emitChange('insert', [v], rest)
        }
        if (rest.ok === 0) {
            emitError('save', [v], rest.err)
        }

        return rest
    }


    /**
     * 儲存數據，以主鍵為準更新既有數據，未給之欄位保留；主鍵不存在且option.autoInsert為true(預設)時改為插入
     *
     * 註: 回傳陣列恆與輸入等長，輸入單一物件亦回傳長度1之陣列
     * 註: n為主鍵命中筆數，值為0或1，命中(不論內容有無變更)或經插入而產生皆為1；nInserted與nModified恆同時出現
     * 註: [內容相同]之判定基準為將待寫入物件合併進現值後與現值相同，非待寫入物件與現值全等，
     * 故只給部份欄位且該些欄位值皆與現值相同時，合併結果等於現值，nModified為0
     * 註: 本筆失敗不中斷整批，該筆以ok為0並附err回報
     * 註: opt.autoGenPk為true(預設)時未帶有效主鍵者自動產生，為false時未帶有效主鍵即reject且同批皆不寫入
     * 註: 主鍵值依主鍵欄型別正規化(字串類主鍵欄之123即'123')；型別不符(整數類主鍵欄給'012'等)屬呼叫端未履行契約，整批reject且同批皆不寫入
     * 註: DATE、DATEONLY欄之值換算後年份須在1–9999，範圍外或無效者(如new Date(8.64e15)、new Date(NaN))屬呼叫端未履行契約，整批reject且同批皆不寫入(含autoInsert為false而不會寫入者；不降為該筆ok為0)，
     * 訊息為invalid data: date is not a valid time in years 1-9999 at [data[索引].欄名]，不回顯值，兩方言一致於送出前拒收，規則同insert
     * 註: 主鍵與數據之字串須良構(孤立代理碼元、主鍵之NUL；規則同insert, 第66項, R13)，但不合格者為該筆失敗而非整批reject：該筆ok為0並附err(invalid id: 原因 at [data[索引].主鍵欄]或invalid data: 原因 at [data[索引].路徑]，不回顯值)，
     * 不讀取資料庫、不寫入(含autoInsert為false)，同批其他筆照常處理；不隨opt.autoGenPk改變。全部筆數皆不合格時不連線
     *
     * @memberOf WOrmReladb
     * @param {Object|Array} data 輸入數據物件或陣列
     * @param {Object} [option={}] 輸入設定物件，預設為{}
     * @param {Object} [option.instance=null] 輸入實例instance物件，預設為null
     * @param {Object} [option.transaction=null] 輸入交易(transaction)物件，預設為null
     * @param {boolean} [option.autoInsert=true] 輸入是否於儲存時發現原本無數據，則自動改以插入處理，預設為true
     * @returns {Promise} 回傳Promise，resolve回傳與輸入等長之儲存結果陣列[{n,nInserted,nModified,ok}]，reject回傳錯誤訊息
     */
    async function save(data, option = {}) {

        //check
        if (!iseobj(data) && !isearr(data)) {
            return []
        }

        //instance
        let instance = get(option, 'instance', null)

        //transaction
        let transaction = get(option, 'transaction', null)

        //check
        if (opt.useEncryption && dialect === 'sqlite') {
            if (transaction !== null) {
                diag('@journeyapps/sqlcipher can not support transaction.')
            }
        }

        //autoInsert
        let autoInsert = get(option, 'autoInsert', true)

        //isErr, res
        let isErr = false
        let res = null

        //si
        let si = instance

        //inited, 本次呼叫是否已進入initSequelize(僅此時finally才關閉sequelize; 預檢失敗於其前者不得關閉呼叫端先前持有之連線)
        let inited = false

        try {

            //check, 實例已關閉屬整批性錯誤, 於入口快速失敗
            procClosed(instance)

            //chkWf, 主鍵與資料之字串良構(孤立代理碼元、主鍵之NUL)逐筆檢核(R13): 先於連線、model載入與任何讀取、寫入、刪除, 檢查之物件即cloneDeep後之複本且其後只使用該複本;
            //不合格者為該筆失敗(ok為0附err, 同批其他筆照常), bads為逐筆訊息(合格者為null); 不隨autoGenPk改變
            let bads = chkWf(isarr(data) ? data : [data], opt.pk, true)
            let nOk = bads.filter((v) => v === null).length

            //initSequelize, 內部自動初始化須在try內: 其例外(如驅動缺失)與si.err同屬整批性錯誤, 方能於reject前發出error事件, 已建構之sequelize亦由finally關閉; 全部不合格者無須連線
            if (instance === null && nOk > 0) {
                inited = true
                si = await initSequelize()
            }

            //check
            if (si !== null && si.err) {
                throw si.err
            }

            //md
            let md = si === null ? null : si.mds[opt.cl]

            //check
            if (!isarr(data)) {
                data = [data]
            }

            //procPk, 須於任何寫入之前一次完成, 令autoGenPk為false而拋錯時同批之有效筆數亦不會被寫入
            data = map(data, (v, k) => {
                return bads[k] !== null ? v : procPk(md, v, k)
            })

            //normPks, 主鍵值依主鍵欄型別正規化後寫回, 型別不符者整批reject且不寫入任何一筆
            normPks(md, data, bads)

            //chkDates, DATE、DATEONLY欄之值須為年份在1–9999之有效時間, 範圍外或無效者整批reject且不寫入任何一筆(含autoInsert為false而不會寫入者; 與條件側共用範圍判定, 見R12)
            chkDates(md, data, bads)

            //ks, model之欄位名, 供合併後比對時濾除非欄位之鍵
            let ks = keys(get(md, 'rawAttributes', {}))

            //pmSeries
            res = await pmSeries(data, async(v, k) => {

                //check, 字串不合格之筆為該筆失敗: 不讀取資料庫、不寫入, 同批其他筆照常
                if (bads[k] !== null) {
                    emitError('save', [v], bads[k])
                    return {
                        n: 1,
                        nInserted: 0,
                        nModified: 0,
                        ok: 0,
                        err: bads[k],
                    }
                }

                return saveOne(md, v, autoInsert, transaction, ks, k)
            })

        }
        catch (err) {
            isErr = true
            res = outErr(err, si) //外來值經diag.err, 本套件之固定詞彙原樣; reject之值與error事件同源
        }
        finally {

            //closeSequelize
            if (instance === null && inited) { //內部自動初始化得close
                await closeSequelize('save')
            }

        }

        //check
        if (isErr) {
            emitError('save', data, res)
            return Promise.reject(res)
        }

        //emit, 須於結果定案後發出
        emitChange('save', data, res)

        return res
    }


    /**
     * 刪除數據
     *
     * 註: 回傳陣列恆與輸入等長，各筆之n與nDeleted皆為主鍵命中筆數，值為0或1，未命中為0且屬正常結果
     * 註: 判斷某筆是否真的被刪除一律以nDeleted為準
     * 註: 未帶有效主鍵者為該筆之輸入問題，回ok為0與err且不送查詢條件(以免無效值被轉為null而誤中其他數據)，
     * 不中斷其餘筆數，以此與[主鍵未命中]之ok為1區辨；主鍵值與主鍵欄型別不符者(整數類主鍵欄給'012'等)亦同
     * 註: 主鍵值依主鍵欄型別正規化後刪除(字串類主鍵欄之123即'123')，正規化值寫回該筆數據
     * 註: 本函數不受opt.autoGenPk影響，於任一設定下皆不補值
     * 註: 主鍵含孤立代理碼元或NUL者(含new String包裝者；UTF-8存放之資料庫會把孤立代理碼元改寫為U+FFFD而刪到他筆)為該筆失敗，ok為0並附err(invalid id: 原因 at [data[索引].主鍵欄]，不回顯值)，
     * 不送查詢，同批其他筆照常；只檢主鍵，其他欄位不檢(第66項, R13)；全部筆數皆不合格時不連線。其他無效主鍵之err亦不回顯主鍵值(invalid data[索引].主鍵欄…)
     *
     * @memberOf WOrmReladb
     * @param {Object|Array} data 輸入數據物件或陣列，會查找各數據的opt.pk值，有存在者就刪除
     * @param {Object} [option={}] 輸入設定物件，預設為{}
     * @param {Object} [option.instance=null] 輸入實例instance物件，預設為null
     * @param {Object} [option.transaction=null] 輸入交易(transaction)物件，預設為null
     * @returns {Promise} 回傳Promise，resolve回傳與輸入等長之刪除結果陣列[{n,nDeleted,ok}]，reject回傳錯誤訊息
     */
    async function del(data, option = {}) {

        //check
        if (!iseobj(data) && !isearr(data)) {
            return []
        }

        //instance
        let instance = get(option, 'instance', null)

        //transaction
        let transaction = get(option, 'transaction', null)

        //check
        if (opt.useEncryption && dialect === 'sqlite') {
            if (transaction !== null) {
                diag('@journeyapps/sqlcipher can not support transaction.')
            }
        }

        //isErr, res
        let isErr = false
        let res = null

        //si
        let si = instance

        //inited, 本次呼叫是否已進入initSequelize(僅此時finally才關閉sequelize; 預檢失敗於其前者不得關閉呼叫端先前持有之連線)
        let inited = false

        try {

            //check, 實例已關閉屬整批性錯誤, 於入口快速失敗
            procClosed(instance)

            //chkWf, 主鍵之字串良構(孤立代理碼元、主鍵之NUL)逐筆檢核(R13): 先於連線、model載入與任何讀取、寫入、刪除, 檢查之物件即cloneDeep後之複本且其後只使用該複本;
            //不合格者為該筆失敗(ok為0附err, 同批其他筆照常), bads為逐筆訊息(合格者為null); 不隨autoGenPk改變
            let bads = chkWf(isarr(data) ? data : [data], opt.pk, false)
            let nOk = bads.filter((v) => v === null).length

            //initSequelize, 內部自動初始化須在try內: 其例外(如驅動缺失)與si.err同屬整批性錯誤, 方能於reject前發出error事件, 已建構之sequelize亦由finally關閉; 全部不合格者無須連線
            if (instance === null && nOk > 0) {
                inited = true
                si = await initSequelize()
            }

            //check
            if (si !== null && si.err) {
                throw si.err
            }

            //md
            let md = si === null ? null : si.mds[opt.cl]

            //check
            if (!isarr(data)) {
                data = [data]
            }

            //pmSeries
            res = await pmSeries(data, async(v, k) => {

                //rest
                let rest = null

                //check, 主鍵字串不合格之筆為該筆失敗: 不送查詢, 同批其他筆照常
                if (bads[k] !== null) {
                    rest = {
                        n: 0,
                        nDeleted: 0,
                        ok: 0,
                        err: bads[k],
                    }
                    emitError('del', [v], rest.err)
                    return rest
                }

                //pkv
                let pkv = get(v, opt.pk)

                //check, 未帶有效主鍵者不補值亦不送查詢, 直接視為該筆無法處理
                if (!isEffPk(pkv)) {

                    //rest
                    rest = {
                        n: 0,
                        nDeleted: 0,
                        ok: 0,
                        err: `invalid data[${k}].${opt.pk}`,
                    }

                    //emit
                    emitError('del', [v], rest.err)

                    return rest
                }

                //normPkOf, 主鍵值與主鍵欄型別不符亦屬未帶有效主鍵, 不送查詢(送出時兩方言皆會把整數欄之'012'轉為12而誤刪)
                let r = normPkOf(md, pkv)
                if (r.kind !== 'ok') {

                    //rest
                    rest = {
                        n: 0,
                        nDeleted: 0,
                        ok: 0,
                        err: `invalid data[${k}].${opt.pk}, type does not match pk column`,
                    }

                    //emit
                    emitError('del', [v], rest.err)

                    return rest
                }

                //寫回正規化值, 令事件之數據與實際送出者一致
                v[opt.pk] = r.v

                //setting
                let setting = {
                    where: { [opt.pk]: r.v },
                }
                if (transaction !== null) {
                    setting.transaction = transaction //destroy的說明寫不需transaction但實際需要 (2020/10/07)
                }

                //destroy, 單一DELETE語句即可取得命中筆數, 不另行findOne以免多一次往返且與刪除間存在競態
                await md.destroy(setting)
                    .then((r) => {

                        //nDel, n與nDeleted皆為主鍵命中筆數, 未命中為0且屬正常結果
                        let nDel = isnum(r) ? r : 0

                        //rest
                        rest = {
                            n: nDel,
                            nDeleted: nDel,
                            ok: 1,
                        }

                    })
                    .catch((err) => {

                        //check, 連線層錯誤影響後續所有筆數, 依T4屬整批性錯誤, 往外拋而不降為本筆ok為0
                        if (isBatchLevelError(err)) {
                            throw err
                        }

                        //rest, 本筆失敗不中斷整批; err為後端之外來值, 建立時即經diag.mask(R-DIAG), 令回傳值與error事件同源
                        rest = {
                            n: 1,
                            nDeleted: 0,
                            ok: 0,
                            err: diag.mask(getErrMsg(err)),
                        }

                        //emit
                        emitError('del', [v], rest.err)

                    })

                return rest
            })

        }
        catch (err) {
            isErr = true
            res = outErr(err, si) //外來值經diag.err, 本套件之固定詞彙原樣; reject之值與error事件同源
        }
        finally {

            //closeSequelize
            if (instance === null && inited) { //內部自動初始化得close
                await closeSequelize('del')
            }

        }

        //check
        if (isErr) {
            emitError('del', data, res)
            return Promise.reject(res)
        }

        //emit, 須於結果定案後發出
        emitChange('del', data, res)

        return res
    }


    /**
     * 依條件刪除多筆數據(破壞性操作, 須審慎使用, 非必要不用; 刪除單筆請用del), 需與del分開，避免未傳數據導致直接刪除全表
     *
     * 使用須知:
     * 1. 不給條件即刪除全部數據: find未給、為undefined或為{}時, 刪除資料表內全部數據, 不另行確認。給了條件才是部分刪除。
     * 2. 傳了值卻不是條件物件(null、陣列含[]與[{...}]、字串、數值、布林、函數、Date、RegExp、Map、Set、類別實例等)
     *    一律整批reject, 不視為{}, 不刪除任何數據。
     * 3. 條件僅支援系列白名單: 欄位等值{欄:純量}(多欄為AND)、欄位運算子$eq $ne $gt $gte $lt $lte(值為純量)與
     *    $in $nin(值為純量陣列)、文件層$and $or(值為條件物件之非空陣列, 最多999項)。純量為字串、有限數字、布林、有效Date;
     *    null僅可用於欄位等值、$eq、$ne。白名單外之運算子($not $nor $regex $options $exists $where $expr $like $ilike等)、
     *    sequelize之Op Symbol、RegExp值、陣列值(請改用$in)、欄位層$and/$or、子查詢、函數值、Symbol鍵、undefined值、
     *    空的$nin/$and/$or等一律整批reject, 訊息為invalid find: 原因 at [路徑], 且不刪除任何數據。
     * 4. 與select共用同一檢核與同一條件, select(find)命中之數據即本函數刪除之數據; 建議先以select(find)確認命中範圍再刪除。
     *    欄名須完全等於model之屬性名或資料庫欄名(區分大小寫); 純量須符合欄位型別; 主鍵欄之條件值依主鍵欄型別正規化。
     * 5. 條件值若來自可能為undefined、null或空陣列之變數, 呼叫端須先自行判斷; undefined值會reject而非被忽略,
     *    空的$nin會reject, $in:[]則為恆假(刪除0筆)。
     * 6. 語意差異(SQL式): $ne/$nin不命中欄位為NULL之數據(三值邏輯)。需明確表達時可用可攜寫法:
     *    排除NULL為{$and:[{f:{$ne:null}},{f:{$ne:x}}]}, 含NULL為{$or:[{f:null},{f:{$ne:x}}]}; 字串比對之大小寫依資料庫定序。
     *
     * 註: n與nDeleted皆為實際刪除筆數，兩者恆相等，n不得取全表筆數
     * 註: 條件無命中時回{n:0,nDeleted:0,ok:1}，不視為錯誤
     *
     * @memberOf WOrmReladb
     * @param {Object} [find={}] 輸入刪除條件物件(系列白名單, 見使用須知)，不給予find則代表刪除全部數據
     * @param {Object} [option={}] 輸入設定物件，預設為{}
     * @param {Object} [option.instance=null] 輸入實例instance物件，預設為null
     * @param {Object} [option.transaction=null] 輸入交易(transaction)物件，預設為null
     * @returns {Promise} 回傳Promise，resolve回傳刪除結果{n,nDeleted,ok}，reject回傳錯誤訊息
     */
    async function delAll(find = {}, option = {}) {

        //instance
        let instance = get(option, 'instance', null)

        //transaction
        let transaction = get(option, 'transaction', null)

        //check
        if (opt.useEncryption && dialect === 'sqlite') {
            if (transaction !== null) {
                diag('@journeyapps/sqlcipher can not support transaction.')
            }
        }

        //isErr, res
        let isErr = false
        let res = null

        //si
        let si = instance

        //inited, 本次呼叫是否已進入initSequelize(僅此時finally才關閉sequelize; 預檢失敗於其前者不得關閉呼叫端先前持有之連線)
        let inited = false

        try {

            //check, 實例已關閉屬整批性錯誤, 於入口快速失敗
            procClosed(instance)

            //chkFindShape, 條件形狀不合法者(系列白名單, R11)於載入model與建立sequelize實例之前即拋錯: 不連線、不送出任何SQL、不刪除任何數據; 其後只使用檢核後之複本
            let fd = chkFindShape(find)

            //initSequelize, 內部自動初始化須在try內: 其例外(如驅動缺失)與si.err同屬整批性錯誤, 方能於reject前發出error事件, 已建構之sequelize亦由finally關閉
            if (instance === null) {
                inited = true
                si = await initSequelize()
            }

            //check
            if (si.err) {
                throw si.err
            }

            //md
            let md = si.mds[opt.cl]

            //useFind, 與select共用cvFind(對複本再檢核一次, 冪等), 令刪除之數據即select命中之數據; 翻譯層不合法者於此拋錯而不刪除任何數據
            let useFind = cvFind(fd, getFindCtx(md))

            //chkScopeForDelAll, model之預設scope會使刪除對象與select分歧者拒絕
            chkScopeForDelAll(md)

            //setting
            let setting = {
                where: useFind,
            }
            if (transaction !== null) {
                setting.transaction = transaction //destroy的說明寫不需transaction但實際需要 (2020/10/07)
            }

            //destroy, n與nDeleted皆為實際刪除筆數, 條件無命中時為0且屬正常結果
            let r = await md.destroy(setting)
            let nDel = isnum(r) ? r : 0

            //res
            res = {
                n: nDel,
                nDeleted: nDel,
                ok: 1,
            }

        }
        catch (err) {
            isErr = true
            res = outErr(err, si) //外來值經diag.err, 本套件之固定詞彙原樣; reject之值與error事件同源
        }
        finally {

            //closeSequelize
            if (instance === null && inited) { //內部自動初始化得close
                await closeSequelize('delAll')
            }

        }

        //check
        if (isErr) {
            emitError('delAll', null, res)
            return Promise.reject(res)
        }

        //emit, 須於結果定案後發出
        emitChange('delAll', null, res)

        return res
    }

    /**
     * 由指定資料庫生成各資料表的models資料

     *
     * 診斷輸出(sql紀錄與建表、外鍵查詢失敗)經本套件單一出口：預設承接建構時之opt.logging、opt.logDetail、opt.funLog，標籤加前綴'genModelsByDB '(如'genModelsByDB sql'、'genModelsByDB tables catch')，option內給予者(型別正確)覆寫單次呼叫；連線字串、帳號、密碼與sqlite檔案路徑(含opt.url、option.username、option.password、option.storage)於細節皆以***遮罩
     *
     * reject與同步拋出之值為w-auto-sequelize所給(其固定詞彙如need username原樣；後端錯誤中單次呼叫之憑證已由w-auto-sequelize遮罩)
     *
     * @memberOf WOrmReladb
     * @param {Object} [option={}] 輸入設定物件，預設{}
     * @param {String} [option.storage='./worm.db'] 輸入sqlite資料庫檔案位置字串，預設'./worm.db'
     * @param {String} [option.db='worm'] 輸入資料庫名稱字串，預設'worm'
     * @param {String} [option.username='username'] 輸入使用者名稱字串，預設'username'
     * @param {String} [option.password='password'] 輸入密碼字串，預設'password'
     * @param {String} [option.dialect='mssql'] 輸入資料庫種類字串，預設'mssql'，可選'mssql', 'sqlite', 'mysql', 'mariadb', 'postgres'
     * @param {String} [option.fdModels='./models'] 輸入models儲存的資料夾名稱字串，預設'./models'
     * @param {String} [option.host='localhost'] 輸入連線主機host位址字串，預設'localhost'
     * @param {Integer} [option.port=1433] 輸入連線主機port整數，預設1433
     * @param {Boolean} [option.logging] 輸入是否紀錄實際執行的sql指令布林值，預設承接建構時之opt.logging，型別錯誤者沿用建構時之設定。為true時預設只輸出敘述種類(如'genModelsByDB sql { stmt: 'SELECT' }')，sql全文需搭配logDetail或funLog
     * @param {Boolean} [option.logDetail] 輸入是否於console輸出遮罩後之細節布林值，預設承接建構時之opt.logDetail，型別錯誤者沿用建構時之設定。為true時console輸出(label, 細節)：sql全文以console.log('genModelsByDB sql', sql全文)、建表或外鍵查詢失敗之錯誤物件副本以console.error('genModelsByDB tables catch', 錯誤)等，有洩漏個資之風險，僅供除錯使用，連線字串、帳密與sqlite檔案路徑仍遮罩
     * @param {Function} [option.funLog] 輸入診斷輸出函數(label, safe, detail)，預設承接建構時之opt.funLog，非函數者沿用建構時之設定。給予時診斷輸出只交給此函數(不寫console)：label為'genModelsByDB sql'、'genModelsByDB foreign keys catch'、'genModelsByDB tables catch'，safe為不含個資之摘要(如{ stmt: 'SELECT' }、{ errType: 'SequelizeConnectionError', errCode: 'SQLITE_CANTOPEN' })，detail為完整細節(帳密與sqlite檔案路徑已遮罩為***)
     * @returns {Promise} 回傳Promise，resolve回傳產生的models資料{ tables, foreignKeys }，reject回傳錯誤(缺必填等w-auto-sequelize自身之檢查為固定字串如need username；建表清單、描述表、外鍵查詢、連線、建立資料夾與寫檔失敗為後端或fs之錯誤，其中單次呼叫之憑證已遮罩)
     */
    function genModelsByDB(option = {}) {

        //default
        let def = {
            username: 'username',
            password: 'password',
            dialect: 'mssql',
            host: 'localhost',
            port: 1433,
            storage: './worm.db',
        }

        //merge
        option = {
            ...def,
            ...option,
        }

        //database
        if (!option.db) {
            option.db = 'worm'
        }
        option.database = opt.db

        //directory
        if (!option.fdModels) {
            option.fdModels = './models'
        }
        option.directory = opt.fdModels

        //storage
        let storageCallRaw = option.storage
        option.storage = path.resolve(option.storage)

        //optLogCall, 診斷設定(R17-7): 預設承接建構時之快照optLog(opt.logging、opt.logDetail、opt.funLog), option內型別正確者覆寫單次呼叫
        let optLogCall = mergeOptLog(optLog, option)

        //diagCall, 單次呼叫之出口: 本套件之宣告cfg(建構時之連線字串與帳密)以diag.with併入單次呼叫之帳號、密碼與sqlite檔案路徑(dspec §6.2第7點)
        let pathsCall = option.dialect === 'sqlite' && isestr(storageCallRaw) ? [storageCallRaw, path.normalize(storageCallRaw), option.storage] : []
        let diagCall = getDiag(optLogCall, cfg).with({ username: option.username, password: option.password, paths: pathsCall })

        //funLogForAuto, w-auto-sequelize一律給funLog(其輸出不再自寫console), 經此轉入本套件出口, 使輸出遵守logDetail與funLog, 細節(w-auto-sequelize已以單次之帳密遮罩之副本)再經diagCall遮罩(含opt.url);
        //label只用固定詞彙: w-auto-sequelize之'sql'、'foreign keys catch'、'tables catch'加前綴'genModelsByDB ', 其他(未知)為'genModelsByDB diag'; sql走stdout, 兩個錯誤標籤走stderr(level 'error', 舊版為console.error)
        let labels = {
            'sql': 'genModelsByDB sql',
            'foreign keys catch': 'genModelsByDB foreign keys catch',
            'tables catch': 'genModelsByDB tables catch',
        }
        let funLogForAuto = (label, safe, detail) => {
            let lb = typeof label === 'string' && Object.prototype.hasOwnProperty.call(labels, label) ? labels[label] : 'genModelsByDB diag'
            diagCall(lb, safe, detail === undefined ? undefined : () => detail, label === 'sql' ? 'log' : 'error')
        }

        //WAutoSequelize, 其reject與同步拋出之值不再經本套件遮罩: w-auto-sequelize已於其catch站點以單次呼叫之帳密經diag.err(建構時之帳密不傳給它, 不會出現於其錯誤),
        //且其固定詞彙之reject(need username等)若再經遮罩, 帳密恰為其子字串時會被切碎(dspec D3)
        return WAutoSequelize({
            ...option,
            logging: optLogCall.logging,
            logDetail: optLogCall.logDetail,
            funLog: funLogForAuto,
        })

    }


    /**
     * 由資料表物件生成各資料表的models資料
     *
     * 診斷輸出(產生之檔案路徑、寫檔失敗之錯誤)經本實例之單一出口(遵守opt.logDetail、opt.funLog)：標籤為'generate file:'、'save file catch'，預設只輸出{ name, type }與錯誤類別、代碼，路徑與錯誤原文為細節；寫檔失敗只記錄、不拋錯
     *
     * @memberOf WOrmReladb
     * @param {String} [fd='./models'] 輸入models儲存的資料夾名稱字串，預設'./models'
     * @param {Object} [tabs={}] 輸入各資料表物件，預設{}
     * @param {Object} [opt={}] 輸入設定物件，預設{}
     * @param {String} [opt.type='js'] 輸入資料表類型字串，預設'js'
     */
    function genModelsByTabs(...input) {
        _genModelsByTabs(input[0], input[1], input[2], diag)
    }


    //bind
    ee.createStorage = createStorage
    ee.genModelsByDB = genModelsByDB
    ee.genModelsByTabs = genModelsByTabs
    ee.init = initSequelize
    // ee.close = () => {
    //     return closeSequelize('external')
    // }
    ee.genTransaction = genTransaction


    //snap, 數據於呼叫當下即同步複製(R13): useStable時各函數經佇列延後執行(pmQueue於執行前delay), 若於函數內才複製,
    //呼叫端於呼叫後立即改動原數據(如重用同一陣列)會影響實際檢核與寫入之值; 函數內只使用此複本, 檢核與寫入皆作用於同一份複本
    //複製拋錯(如getter拋錯)與函數內之其他錯誤同為rejected promise; 其錯誤值為呼叫端數據之getter所拋(外來值), 經diag.err(R-DIAG)
    let snap = (fn) => {
        return function(data, ...rest) {
            let d
            try {
                d = cloneDeep(data)
            }
            catch (err) {
                return Promise.reject(diag.err(err))
            }
            return fn(d, ...rest)
        }
    }

    if (opt.useStable) {
        //用佇列(同時最大執行數1且先進先執行)處理高併發之情形
        //若沒管控:
        //mssql就是先執行先贏
        //Sqlite有機會無法儲存, 或是出錯[Error: SQLITE_MISUSE: bad parameter or other API misuse]或[Error: SQLITE_MISUSE: Database handle is closed]
        ee.select = function() {
            return pmq(select, ...arguments)
        }
        ee.selectByPk = function() {
            return pmq(selectByPk, ...arguments)
        }
        ee.insert = snap((...args) => {
            return pmq(insert, ...args)
        })
        ee.insertBulk = snap((...args) => {
            return pmq(insertBulk, ...args)
        })
        ee.save = snap((...args) => {
            return pmq(save, ...args)
        })
        ee.del = snap((...args) => {
            return pmq(del, ...args)
        })
        ee.delAll = function() {
            return pmq(delAll, ...arguments)
        }
    }
    else {
        ee.select = select
        ee.selectByPk = selectByPk
        ee.insert = snap(insert)
        ee.insertBulk = snap(insertBulk)
        ee.save = snap(save)
        ee.del = snap(del)
        ee.delAll = delAll
    }

    return ee
}
